mock_response = MagicMock(spec=httpx.Response)
mock_response.status_code = 200
mock_response.text = _SAMPLE_ATOM_XML

mock_client = MagicMock(spec=httpx.AsyncClient)
mock_client.get = AsyncMock(return_value=mock_response)
mock_client.aclose = AsyncMock()

with patch("app.services.arxiv_fetch.httpx.AsyncClient",
           return_value=mock_client):
    papers = asyncio.run(fetch_arxiv_papers("all:test", max_results=1))

3.4 Mock 限流器

class _NoopSlot:
    """替换 _global_request_slot 的无操作异步上下文管理器"""
    async def __aenter__(self): return None
    async def __aexit__(self, *args): pass

@pytest.fixture(autouse=True)
def _mock_rate_limit(self):
    patcher = patch.object(
        app.services.arxiv_fetch, "_global_request_slot",
        return_value=_NoopSlot())
    patcher.start()
    yield
    patcher.stop()
